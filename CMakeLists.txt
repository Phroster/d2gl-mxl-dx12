cmake_minimum_required(VERSION 3.24)
project(MXLSmoothMotionDX12 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Visual Studio 2022 and -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mxl_diagnostics STATIC experimental/dx12/diagnostics.cpp experimental/dx12/audio_diagnostics.cpp experimental/dx12/input_profile.cpp)
target_include_directories(mxl_diagnostics PUBLIC experimental/dx12 PRIVATE d2gl/d2gl/vendor/include)
target_compile_definitions(mxl_diagnostics PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(mxl_diagnostics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/d2gl/d2gl/vendor/lib/detours.lib" dxguid user32 psapi)
add_executable(dx12_input_profile_test experimental/tests/input_profile_test.cpp)
target_link_libraries(dx12_input_profile_test PRIVATE mxl_diagnostics)
target_compile_definitions(dx12_input_profile_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SKIP_GLSLANG_INSTALL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)

set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL ON CACHE BOOL "" FORCE)
add_subdirectory(external/DiligentCore/ThirdParty/glslang EXCLUDE_FROM_ALL)
add_subdirectory(external/DiligentCore/ThirdParty/SPIRV-Cross EXCLUDE_FROM_ALL)
add_library(mxl_dx12_shader STATIC experimental/dx12/shader_bridge.cpp)
target_include_directories(mxl_dx12_shader PUBLIC experimental/dx12
  PRIVATE external/DiligentCore/ThirdParty/glslang external/DiligentCore/ThirdParty/SPIRV-Cross)
target_compile_definitions(mxl_dx12_shader PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mxl_dx12_shader PUBLIC glslang SPIRV glslang-default-resource-limits spirv-cross-hlsl spirv-cross-glsl d3dcompiler)
add_executable(dx12_shader_test experimental/tests/shader_test.cpp)
target_link_libraries(dx12_shader_test PRIVATE mxl_dx12_shader)
add_executable(dx12_timing_test d2gl/d2gl/src/mxl_smoothing.c)
target_compile_definitions(dx12_timing_test PRIVATE MXL_SMOOTHING_TEST _CRT_SECURE_NO_WARNINGS)
target_link_libraries(dx12_timing_test PRIVATE advapi32)
add_executable(dx12_preset_test experimental/tests/preset_test.cpp)
target_link_libraries(dx12_preset_test PRIVATE mxl_dx12_shader)
target_compile_definitions(dx12_shader_test PRIVATE MXL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_library(mxl_dx12_device STATIC experimental/dx12/device.cpp)
target_include_directories(mxl_dx12_device PUBLIC experimental/dx12)
target_link_libraries(mxl_dx12_device PUBLIC d3d12 dxgi dxguid mxl_diagnostics)
add_executable(dx12_diagnostics_test experimental/tests/diagnostics_test.cpp)
target_link_libraries(dx12_diagnostics_test PRIVATE mxl_dx12_device dsound)

  target_include_directories(dx12_mpq_blast PRIVATE external/blast)

file(GLOB_RECURSE MXL_CORE_SOURCES CONFIGURE_DEPENDS d2gl/d2gl/src/*.cpp)

set(MXL_IMGUI_SOURCES
  "${MXL_IMGUI_DIR}/imgui.cpp" "${MXL_IMGUI_DIR}/imgui_draw.cpp"
  "${MXL_IMGUI_DIR}/imgui_widgets.cpp" "${MXL_IMGUI_DIR}/imgui_tables.cpp"
  "${MXL_IMGUI_DIR}/imgui_impl_win32.cpp" "${MXL_IMGUI_DIR}/imgui_impl_dx12.cpp")

  "${MXL_IMGUI_DIR}/imgui.cpp" "${MXL_IMGUI_DIR}/imgui_draw.cpp"
  "${MXL_IMGUI_DIR}/imgui_widgets.cpp" "${MXL_IMGUI_DIR}/imgui_tables.cpp" "${MXL_IMGUI_DIR}/imgui_impl_dx12.cpp")
target_include_directories(dx12_menu_test PRIVATE experimental/dx12 d2gl/d2gl/vendor/include)
target_compile_definitions(dx12_menu_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS MXL_BUILD_DIR="${CMAKE_BINARY_DIR}")
target_link_libraries(dx12_menu_test PRIVATE mxl_dx12_device d3dcompiler)
foreach(renderer glide3x ddraw)
  file(GLOB_RECURSE WRAPPER_SOURCES CONFIGURE_DEPENDS "d2gl/${renderer}/src/*.cpp")
  add_library(${renderer} SHARED ${MXL_CORE_SOURCES} ${WRAPPER_SOURCES} ${MXL_IMGUI_SOURCES}
    d2gl/d2gl/src/mxl_smoothing.c experimental/dx12/glslang_compat.cpp experimental/dx12/fps_menu.cpp
    "d2gl/${renderer}/${renderer}.rc")
  target_include_directories(${renderer} PRIVATE experimental/dx12 d2gl/d2gl/src
    d2gl/d2gl/vendor/include "d2gl/${renderer}/src")
  target_compile_definitions(${renderer} PRIVATE WIN32 UNICODE _UNICODE _WINDOWS _USRDLL GLEW_STATIC NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  target_compile_options(${renderer} PRIVATE /EHsc /fp:fast /arch:SSE2)
  target_link_libraries(${renderer} PRIVATE mxl_dx12_backend
    "${CMAKE_CURRENT_SOURCE_DIR}/d2gl/d2gl/vendor/lib/detours.lib"
    advapi32 version winmm user32 gdi32 shell32 imm32 dwmapi)
  target_link_options(${renderer} PRIVATE /PDBALTPATH:%_PDB%)
endforeach()
target_compile_definitions(glide3x PRIVATE GLIDE3X_EXPORTS)
