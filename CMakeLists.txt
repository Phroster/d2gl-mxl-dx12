cmake_minimum_required(VERSION 3.24)
project(MXLSmoothMotionDX12 VERSION 1.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Visual Studio 2022 and -A Win32.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mxl_diagnostics STATIC src/dx12/diagnostics.cpp src/dx12/audio_diagnostics.cpp src/dx12/sound_probe.cpp src/dx12/sound_cancel.cpp src/dx12/asset_probe.cpp src/dx12/tile_cache.cpp src/dx12/archive_hash_cache.cpp src/dx12/input_profile.cpp src/dx12/reveal_probe.cpp src/dx12/auto_reveal.cpp)
target_include_directories(mxl_diagnostics PUBLIC src/dx12 PRIVATE d2gl/d2gl/vendor/include)
target_compile_definitions(mxl_diagnostics PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(mxl_diagnostics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/d2gl/d2gl/vendor/lib/detours.lib" dxguid user32 psapi advapi32)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
set(ENABLE_SPVREMAPPER OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SKIP_GLSLANG_INSTALL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_MSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_SKIP_INSTALL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL ON CACHE BOOL "" FORCE)
include(cmake/Dependencies.cmake)
add_library(mxl_dx12_shader STATIC src/dx12/shader_bridge.cpp)
target_include_directories(mxl_dx12_shader PUBLIC src/dx12
  PRIVATE "${mxl_glslang_SOURCE_DIR}" "${mxl_spirv_cross_SOURCE_DIR}")
target_compile_definitions(mxl_dx12_shader PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mxl_dx12_shader PUBLIC glslang SPIRV glslang-default-resource-limits spirv-cross-hlsl spirv-cross-glsl d3dcompiler)
add_library(mxl_dx12_device STATIC src/dx12/device.cpp)
target_include_directories(mxl_dx12_device PUBLIC src/dx12)
target_link_libraries(mxl_dx12_device PUBLIC d3d12 dxgi dxguid mxl_diagnostics)
add_library(mxl_dx12_backend STATIC src/dx12/gl_api.cpp src/dx12/draw.cpp src/dx12/mipmap.cpp)
target_include_directories(mxl_dx12_backend PUBLIC src/dx12 d2gl/d2gl/vendor/include)
target_compile_definitions(mxl_dx12_backend PUBLIC GLEW_STATIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mxl_dx12_backend PUBLIC mxl_dx12_device mxl_dx12_shader)
file(GLOB_RECURSE MXL_CORE_SOURCES CONFIGURE_DEPENDS d2gl/d2gl/src/*.cpp)
set(MXL_IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/d2gl/d2gl/vendor/include/imgui")
set(MXL_IMGUI_SOURCES
  "${MXL_IMGUI_DIR}/imgui.cpp" "${MXL_IMGUI_DIR}/imgui_draw.cpp"
  "${MXL_IMGUI_DIR}/imgui_widgets.cpp" "${MXL_IMGUI_DIR}/imgui_tables.cpp"
  "${MXL_IMGUI_DIR}/imgui_impl_win32.cpp" "${MXL_IMGUI_DIR}/imgui_impl_dx12.cpp")
foreach(renderer glide3x ddraw)
  file(GLOB_RECURSE WRAPPER_SOURCES CONFIGURE_DEPENDS "d2gl/${renderer}/src/*.cpp")
  add_library(${renderer} SHARED ${MXL_CORE_SOURCES} ${WRAPPER_SOURCES} ${MXL_IMGUI_SOURCES}
    d2gl/d2gl/src/mxl_smoothing.c src/dx12/glslang_compat.cpp src/dx12/fps_menu.cpp
    "d2gl/${renderer}/${renderer}.rc")
  target_include_directories(${renderer} PRIVATE src/dx12 d2gl/d2gl/src
    d2gl/d2gl/vendor/include "d2gl/${renderer}/src")
  target_compile_definitions(${renderer} PRIVATE WIN32 UNICODE _UNICODE _WINDOWS _USRDLL GLEW_STATIC NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  target_compile_options(${renderer} PRIVATE /EHsc /fp:fast /arch:SSE2)
  target_link_libraries(${renderer} PRIVATE mxl_dx12_backend
    "${CMAKE_CURRENT_SOURCE_DIR}/d2gl/d2gl/vendor/lib/detours.lib"
    advapi32 version winmm user32 gdi32 shell32 imm32 dwmapi)
  target_link_options(${renderer} PRIVATE /PDBALTPATH:%_PDB%)
endforeach()
target_compile_definitions(glide3x PRIVATE GLIDE3X_EXPORTS)
target_compile_definitions(ddraw PRIVATE DDRAW_EXPORTS)
target_sources(ddraw PRIVATE d2gl/ddraw/ddraw.def)

option(MXL_BUILD_TESTS "Build the native verification programs" OFF)
if(MXL_BUILD_TESTS)
  enable_testing()
  include(cmake/Tests.cmake)
endif()
